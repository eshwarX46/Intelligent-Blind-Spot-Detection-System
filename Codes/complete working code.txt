// Short-range prototype thresholds; the base limits apply up to 30 km/h.
float dangerLimit() {
  if (speedKmh > 60) return 8.0f;
  if (speedKmh > 30) return 6.0f;
  return DANGER_DISTANCE;
}

float warningLimit() {
  if (speedKmh > 60) return 15.0f;
  if (speedKmh > 30) return 12.0f;
  return WARNING_DISTANCE;
}

// Read the potentiometer with light smoothing.
void updateVehicleSpeed() {
  if (millis() - lastSpeedTime < SPEED_INTERVAL) return;
  lastSpeedTime = millis();

  uint32_t sum = 0;
  for (int i = 0; i < 8; i++) {
    sum += analogRead(SPEED_POT_PIN);
  }
  int adc = sum / 8;
  int target = map(adc, 0, 4095, 0, 100);
  target = constrain(target, 0, 100);
  // Fast enough for manual adjustment, without abrupt flicker.
  speedKmh = (speedKmh * 3 + target + 2) / 4;
}

// ========================================
// MEDIAN FILTER
// ========================================
float calculateMedian(uint8_t zoneIndex) {
  DetectionZone &zone = zones[zoneIndex];

  if (zone.count == 0) {
    return NAN;
  }

  float values[3];

  for (int i = 0; i < zone.count; i++) {
    values[i] = zone.samples[i];
  }

  for (int i = 0; i < zone.count; i++) {
    for (int j = i + 1; j < zone.count; j++) {

      if (values[j] < values[i]) {
        float temp = values[i];
        values[i] = values[j];
        values[j] = temp;
      }
    }
  }

  if (zone.count == 1) {
    return values[0];
  }

  if (zone.count == 2) {
    return (values[0] + values[1]) / 2.0f;
  }

  return values[1];
}

// ========================================
// ULTRASONIC MEASUREMENT
// ========================================
void updateUltrasonic(uint8_t zoneIndex) {
  DetectionZone &zone = zones[zoneIndex];

  digitalWrite(zone.trigPin, LOW);
  delayMicroseconds(2);

  digitalWrite(zone.trigPin, HIGH);
  delayMicroseconds(10);

  digitalWrite(zone.trigPin, LOW);

  unsigned long duration = pulseIn(
    zone.echoPin,
    HIGH,
    ECHO_TIMEOUT
  );

  float measuredDistance =
      duration * 0.0343f / 2.0f;

  if (duration > 0 &&
      measuredDistance >= MIN_DISTANCE &&
      measuredDistance <= MAX_DISTANCE) {

    zone.samples[zone.index] = measuredDistance;

    zone.index = (zone.index + 1) % 3;

    if (zone.count < 3) {
      zone.count++;
    }

    zone.failures = 0;

    zone.distance = calculateMedian(zoneIndex);

  } else {

    if (zone.failures < 3) {
      zone.failures++;
    }

    if (zone.failures >= 3) {
      zone.distance = NAN;
      zone.count = 0;
      zone.index = 0;
    }
  }
}

// ========================================
// DISTANCE RISK CLASSIFICATION
// ========================================
const char* distanceStatus(float distance) {

  if (isnan(distance)) {
    return "NO ECHO";
  }

  if (distance <= dangerLimit()) {
    return "DANGER";
  }

  if (distance <= warningLimit()) {
    return "WARNING";
  }

  return "SAFE";
}

// ========================================
// OVERALL SYSTEM STATUS
// ========================================
const char* calculateOverallStatus() {

  bool warningFound = false;
  bool unknownFound = false;

  for (int i = 0; i < 3; i++) {

    float distance = zones[i].distance;

    if (isnan(distance)) {
      unknownFound = true;
      continue;
    }

    if (distance <= dangerLimit()) {
      return "DANGER";
    }

    if (distance <= warningLimit()) {
      warningFound = true;
    }
  }

  if (warningFound) {
    return "WARNING";
  }

  if (unknownFound) {
    return "CHECK SENSOR";
  }

  return "SAFE";
}

// ========================================
// LED CONTROL
// ========================================
void updateLEDs() {

  const char* status = calculateOverallStatus();

  bool green = false;
  bool yellow = false;
  bool red = false;

  if (strcmp(status, "SAFE") == 0) {
    green = true;
  }

  else if (strcmp(status, "WARNING") == 0) {
    yellow = true;
  }

  else if (strcmp(status, "DANGER") == 0) {
    red = true;
  }

  else {
    // Blink yellow if a sensor needs checking
    yellow = ((millis() / 500) % 2 == 0);
  }

  digitalWrite(GREEN_LED, green ? HIGH : LOW);
  digitalWrite(YELLOW_LED, yellow ? HIGH : LOW);
  digitalWrite(RED_LED, red ? HIGH : LOW);
}

// ========================================
// BUZZER CONTROL
// ========================================
void setBuzzer(bool on) {
  digitalWrite(BUZZER_PIN,
    (on == BUZZER_ACTIVE_HIGH) ? HIGH : LOW);
}

void updateBuzzer() {
  const char* status = calculateOverallStatus();
  unsigned long now = millis();
  bool sound = false;

  if (strcmp(status, "DANGER") == 0) {
    // Fast intermittent alert: 150ms ON, 150ms OFF
    sound = (now % 300UL) < 150UL;
  } else if (strcmp(status, "WARNING") == 0) {
    // Slow intermittent alert: 150ms ON, 850ms OFF
    sound = (now % 1000UL) < 150UL;
  } else if (strcmp(status, "CHECK SENSOR") == 0) {
    // Short diagnostic alert every three seconds
    sound = (now % 3000UL) < 80UL;
  }

  setBuzzer(sound);
}

// ========================================
// SETUP
// ========================================
void setup() {

  Serial.begin(115200);
  delay(1000);

  // Initialize vehicle-speed input
  pinMode(SPEED_POT_PIN, INPUT);
  analogReadResolution(12);
  speedKmh = map(analogRead(SPEED_POT_PIN), 0, 4095, 0, 100);
  speedKmh = constrain(speedKmh, 0, 100);
  lastSpeedTime = millis();

  // Initialize LED pins
  pinMode(GREEN_LED, OUTPUT);
  pinMode(YELLOW_LED, OUTPUT);
  pinMode(RED_LED, OUTPUT);

  digitalWrite(GREEN_LED, LOW);
  digitalWrite(YELLOW_LED, LOW);
  digitalWrite(RED_LED, LOW);

  // Initialize buzzer
  pinMode(BUZZER_PIN, OUTPUT);
  setBuzzer(false);

  // Initialize ultrasonic pins
  for (int i = 0; i < 3; i++) {

    pinMode(zones[i].trigPin, OUTPUT);
    pinMode(zones[i].echoPin, INPUT);

    digitalWrite(zones[i].trigPin, LOW);
  }

  // Initialize OLED
  Wire.begin(OLED_SDA, OLED_SCL);

  oledReady = display.begin(
    SSD1306_SWITCHCAPVCC,
    OLED_ADDRESS
  );

  if (oledReady) {

    display.clearDisplay();

    display.setTextColor(SSD1306_WHITE);
    display.setTextSize(1);

    display.setCursor(0, 0);
    display.println("INTELLIGENT");
    display.println("BLIND-SPOT SYSTEM");

    display.drawLine(
      0, 19, 127, 19,
      SSD1306_WHITE
    );

    display.setCursor(0, 28);
    display.println("ESP32 CONTROLLER");

    display.setCursor(0, 43);
    display.println("SYSTEM STARTING");

    display.display();

    Serial.println("OLED INITIALIZED");

  } else {

    Serial.println("OLED NOT FOUND");
  }

  // Initialize left radar
  leftRadarSerial.setRxBufferSize(1024);

  leftRadarSerial.begin(
    256000,
    SERIAL_8N1,
    LEFT_RADAR_RX,
    LEFT_RADAR_TX
  );

  // Initialize right radar
  rightRadarSerial.setRxBufferSize(1024);

  rightRadarSerial.begin(
    256000,
    SERIAL_8N1,
    RIGHT_RADAR_RX,
    RIGHT_RADAR_TX
  );

  delay(500);

  leftRadar.begin(leftRadarSerial);
  rightRadar.begin(rightRadarSerial);

  Serial.println("SYSTEM INITIALIZATION COMPLETE");

  lastPingTime = millis();
  lastDisplayTime = millis();
  lastOLEDTime = millis();

  // Network connection instructions appear on OLED until a browser opens the dashboard.
  startWebDashboard();
  showConnectionOLED();
}

// ========================================
// MAIN LOOP
// ========================================
void loop() {

  // Serve browser and live API requests.
  dashboardServer.handleClient();

  // Update radar communication
  leftRadar.read();
  rightRadar.read();

  unsigned long currentTime = millis();

  // Sequential distance measurement
  if (currentTime - lastPingTime >= PING_INTERVAL) {

    lastPingTime = currentTime;

    updateUltrasonic(activeZone);

    activeZone++;

    if (activeZone >= 3) {
      activeZone = 0;
    }
  }

  // Update simulated vehicle speed
  updateVehicleSpeed();

  // Automatic LED operation
  updateLEDs();

  // Automatic buzzer operation
  updateBuzzer();

  // Live OLED update
  if (millis() - lastOLEDTime >= OLED_INTERVAL) {

    lastOLEDTime = millis();

    updateOLED();
  }

  // Display system monitoring information
  if (millis() - lastDisplayTime >= DISPLAY_INTERVAL) {

    lastDisplayTime = millis();

    printDashboard();
  }
}
